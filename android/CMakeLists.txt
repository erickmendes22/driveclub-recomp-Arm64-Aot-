cmake_minimum_required(VERSION 3.24)
project(driveclub_android LANGUAGES C CXX ASM)
set(DRIVECLUB_NATIVE_JOBS 2 CACHE STRING "Concurrent native compiler processes")
set_property(GLOBAL PROPERTY JOB_POOLS driveclub_compile=${DRIVECLUB_NATIVE_JOBS} driveclub_link=1)
set(CMAKE_JOB_POOL_COMPILE driveclub_compile)
set(CMAKE_JOB_POOL_LINK driveclub_link)
include(cmake/VulkanDriver.cmake)
add_library(driveclub_android SHARED src/main/cpp/bridge.cpp src/main/cpp/vulkan_driver_jni.cpp ../launcher/prepare.cpp)
target_compile_features(driveclub_android PRIVATE cxx_std_20)
target_include_directories(driveclub_android PRIVATE ../launcher)
target_link_libraries(driveclub_android PRIVATE driveclub_vulkan_dispatch)
target_link_options(driveclub_android PRIVATE "-Wl,-z,max-page-size=16384")

if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(_dc_pkg_triple aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(_dc_pkg_triple x86_64-linux-android)
else()
    message(FATAL_ERROR "Unsupported Android PKG ABI: ${ANDROID_ABI}")
endif()
add_library(driveclub_pkg_core STATIC IMPORTED)
set_target_properties(driveclub_pkg_core PROPERTIES IMPORTED_LOCATION
    "${CMAKE_CURRENT_SOURCE_DIR}/../out/android-pkg/${_dc_pkg_triple}/release/libdriveclub_pkg_core.a")
add_library(driveclub_pkg SHARED src/main/cpp/pkg_bridge.cpp)
target_compile_features(driveclub_pkg PRIVATE cxx_std_20)
target_link_libraries(driveclub_pkg PRIVATE driveclub_pkg_core log dl m)
target_link_options(driveclub_pkg PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,-z,max-page-size=16384)

if(ANDROID_ABI STREQUAL "arm64-v8a")
    add_library(bbcpu SHARED IMPORTED)
    set_target_properties(bbcpu PROPERTIES IMPORTED_LOCATION
        "${CMAKE_CURRENT_SOURCE_DIR}/../out/android-fex/build/libbbcpu.so")
    target_include_directories(driveclub_android PRIVATE ../src)
    target_link_libraries(driveclub_android PRIVATE bbcpu)

    set(BB_LTO OFF CACHE BOOL "" FORCE)
    set(BB_PGO off CACHE STRING "" FORCE)
    add_subdirectory(../gpu gpu)
    file(GLOB runtime_sources CONFIGURE_DEPENDS ../src/runtime*.c ../third_party/LibAtrac9/C/src/*.c)
    add_library(driveclub_runtime STATIC ../src/probe.c ../src/vulkan_smoke.c ${runtime_sources})
    set_target_properties(driveclub_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_features(driveclub_runtime PRIVATE c_std_11)
    target_compile_definitions(driveclub_runtime PRIVATE DRIVECLUB_ANDROID_RUNTIME=1)
    target_compile_options(driveclub_runtime PRIVATE -O2 -ffunction-sections -fdata-sections)
    target_include_directories(driveclub_runtime PRIVATE .. ../src ../third_party/SDL3-mingw/x86_64-w64-mingw32/include)
    add_library(driveclub_game SHARED src/main/cpp/runtime_bridge.cpp src/main/cpp/vulkan_driver_jni.cpp)
    target_compile_features(driveclub_game PRIVATE cxx_std_20)
    target_compile_definitions(driveclub_game PRIVATE DRIVECLUB_HAS_RUNTIME=1 DRIVECLUB_ANDROID_RUNTIME=1)
    target_include_directories(driveclub_game PRIVATE .. ../src ../launcher)
    target_link_libraries(driveclub_game PRIVATE driveclub_runtime bbgpu bbcpu android log aaudio dl m)
    target_link_options(driveclub_game PRIVATE -Wl,--no-undefined -Wl,--gc-sections -Wl,-z,max-page-size=16384)
endif()
